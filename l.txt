function applySyntaxHighlighting($line) {
    // First handle string literals to prevent false keyword matches inside strings
    $line = preg_replace_callback(
        '/(["\'])(?:\\\\.|(?!\\1).)*\\1/',
        function($matches) {
            return '<span class="string">' . $matches[0] . '</span>';
        },
        $line
    );

    // Then handle PHP keywords, excluding those inside strings
    $line = preg_replace(
        '/\b(abstract|and|array|as|break|callable|case|catch|class|clone|const|continue|declare|default|die|do|echo|else|elseif|empty|enddeclare|endfor|endforeach|endif|endswitch|endwhile|eval|exit|extends|final|finally|fn|for|foreach|function|global|goto|if|implements|include|include_once|instanceof|insteadof|interface|isset|list|match|namespace|new|or|parent|print|private|protected|public|require|require_once|return|static|switch|throw|trait|try|unset|use|var|while|xor|yield|__CLASS__|__DIR__|__FILE__|__FUNCTION__|__LINE__|__METHOD__|__NAMESPACE__|__TRAIT__)\b(?![^<]*<\/span>)/',
        '<span class="keyword">$0</span>',
        $line
    );

    // Comments
    $line = preg_replace(
        '/(\/\/.*|\/\*[\s\S]*?\*\/|#.*)/',
        '<span class="comment">$0</span>',
        $line
    );
    
    // Variables
    $line = preg_replace(
        '/\$[a-zA-Z_\x7f-\xff][a-zA-Z0-9_\x7f-\xff]*/',
        '<span class="variable">$0</span>',
        $line
    );
    
    // Numbers
    $line = preg_replace(
        '/\b\d+\b/',
        '<span class="number">$0</span>',
        $line
    );
    
    // Functions
    $line = preg_replace(
        '/\b[a-zA-Z_\x7f-\xff][a-zA-Z0-9_\x7f-\xff]*(?=\s*\()/',
        '<span class="function">$0</span>',
        $line
    );
    
    // HTML tags
    $line = preg_replace(
        '/&lt;\/?[a-zA-Z][^&]*&gt;/',
        '<span class="html">$0</span>',
        $line
    );
    
    return $line;
}

function highlightSuspiciousPatterns($htmlLine, $originalLine, $patterns) {
    foreach ($patterns as $pattern) {
        if (preg_match('/'.preg_quote($pattern, '/').'/i', $originalLine)) {
            $escapedPattern = preg_quote(htmlspecialchars($pattern, ENT_NOQUOTES, 'UTF-8'), '/');
            $htmlLine = preg_replace(
                "/($escapedPattern)/i",
                '<span class="highlight">$1</span>',
                $htmlLine
            );
        }
    }
    return $htmlLine;
}

function scanFiles($dir, $patterns) {
    $found = false;
    $rii = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($dir));
    
    foreach ($rii as $file) {
        if (!$file->isFile()) continue;
        $path = $file->getPathname();
        if (!preg_match('/\.(php[345]?|phtml?|txt)$/i', $path)) continue;

        $content = @file_get_contents($path);
        $skipMarkers = ['WordPress', '* @version', "if ( ! defined( 'ABSPATH' ) ) {", 'namespace ', '@generated by Composer', '@return bool', '* @param', '@return string', '* Implements', '// Exit if accessed directly', '* @license', '// Set ABSPATH', '* @author', '* @package IXR', '* @"keyword"', '\*======================================================================*/'];
        
        foreach ($skipMarkers as $marker) {
            if (stripos($content, $marker) !== false) {
                continue 2;
            }
        }

        $lines = explode("\n", $content);
        foreach ($lines as $num => $line) {
            foreach ($patterns as $p) {
                if (preg_match('/'.preg_quote($p, '/').'/i', $line)) {
                    $found = true;
                    $urlPath = str_replace($_SERVER['DOCUMENT_ROOT'], '', $path);
                    $realPath = realpath($path);
                    $docRoot = realpath($_SERVER['DOCUMENT_ROOT']);
                    $urlPath = str_replace($docRoot, '', $realPath);
                    $urlFull = (isset($_SERVER['HTTPS']) ? 'https' : 'http') . '://' . $_SERVER['HTTP_HOST'] . $urlPath;
                    $encodedPath = urlencode($path);

                    echo <<<HTML
                    <div class="result-card">
                        <div class="file-path">📁 {$path}</div>
                        <div>
                            <span>Line {$num}</span>
                            <div class="match-line">{$line}</div>
                        </div>
                        <div class="actions">
                            <a href="?src={$encodedPath}" class="btn btn-code" target="_blank">🔍 View Code</a>
                            <a href="{$urlFull}" class="btn btn-view" target="_blank">🌐 View Page</a>
                            <a href="?del={$path}" class="btn btn-delete" onclick="return confirm('Delete this file?');">❌ Delete</a>
                        </div>
                    </div>
HTML;
                    break 2;
                }
            }
        }
    }
    
    if (!$found) {
        echo '<div class="empty-state">
                🍭 No suspicious files found! Your system looks clean.
              </div>';
    }
}
